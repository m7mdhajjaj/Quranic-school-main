┌─────────────────────────────────────────────────────────────────────────────┐
│                    ⚠️ قاعدة التعارض الأساسية                                │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  التعارض يحدث فقط إذا:                                                      │
│  ─────────────────────                                                      │
│  ✅ نفس المعلم (teacherId)                                                  │
│  ✅ نفس التاريخ بالضبط (sessionDate) - ليس فقط نفس اليوم!                  │
│  ✅ تداخل زمني بين startHour/endHour                                       │
│                                                                             │
│  ⚠️ مهم جداً:                                                               │
│  ────────────                                                               │
│  12 يناير (الاثنين) ≠ 19 يناير (الاثنين)                                    │
│  حتى لو كانا نفس اليوم من الأسبوع، لا يوجد تعارض لأن التاريخ مختلف!        │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                          دالة hasTimeOverlap                                │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  تفحص تداخل زمني بين موعدين:                                                │
│                                                                             │
│  function hasTimeOverlap(start1, end1, start2, end2) {                      │
│    const s1 = timeToMinutes(start1);                                        │
│    const e1 = timeToMinutes(end1);                                          │
│    const s2 = timeToMinutes(start2);                                        │
│    const e2 = timeToMinutes(end2);                                          │
│                                                                             │
│    // تداخل إذا: بداية1 < نهاية2 AND نهاية1 > بداية2                        │
│    return s1 < e2 && e1 > s2;                                               │
│  }                                                                          │
│                                                                             │
│  مثال التداخل:                                                              │
│  ─────────────                                                              │
│  موعد 1: 2:00 PM - 3:00 PM                                                  │
│  موعد 2: 2:30 PM - 4:00 PM                                                  │
│  → يوجد تداخل ✗                                                             │
│                                                                             │
│  موعد 1: 2:00 PM - 3:00 PM                                                  │
│  موعد 2: 3:00 PM - 4:00 PM                                                  │
│  → لا يوجد تداخل ✓ (ينتهي الأول عند بداية الثاني)                           │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                          دالة checkTimeConflict                             │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  المدخلات:                                                                   │
│  ─────────                                                                  │
│  {                                                                          │
│    teacherId   : String  - معرف المعلم                                      │
│    sessionDate : Date    - التاريخ المحدد (مطلوب!)                          │
│    startHour   : String  - وقت البداية                                      │
│    endHour     : String  - وقت النهاية                                      │
│    excludeId   : String  - استثناء موعد معين (للتحديث)                      │
│  }                                                                          │
│                                                                             │
│  المخرجات:                                                                   │
│  ─────────                                                                  │
│  {                                                                          │
│    hasConflict : Boolean - هل يوجد تعارض؟                                   │
│    message     : String  - رسالة التعارض                                    │
│    conflictWith: Object  - تفاصيل الموعد المتعارض                           │
│  }                                                                          │
│                                                                             │
│  خطوات الفحص:                                                               │
│  ─────────────                                                              │
│  1. تطبيع التاريخ للبحث (00:00:00 - 23:59:59)                               │
│  2. جلب كل مواعيد المعلم في نفس التاريخ                                     │
│  3. فحص التداخل الزمني مع كل موعد                                           │
│  4. إرجاع تفاصيل التعارض إذا وُجد                                           │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

│    endHour      : "3:30 PM"        (مطلوب) - وقت النهاية                    │
│    teacherId    : "..."            (مطلوب) - معرف المعلم                    │
│    groupId      : "..."            (اختياري) - معرف الحلقة                  │
│    sectionId    : "..."            (اختياري) - معرف المقطع                  │
│    sessionType  : "both"           (اختياري) - نوع الحصة                    │
│    note         : "حلقة الإيمان"   (اختياري) - اسم الحلقة                    │
│    description  : "..."            (اختياري) - وصف إضافي                    │
│  }                                                                          │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                          خطوات الإنشاء                                      │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  1️⃣ التحقق من الصلاحيات:                                                   │
│     • المعلم يمكنه إضافة مواعيد لنفسه فقط                                   │
│     • الإداري يمكنه إضافة لأي معلم                                          │
│                                                                             │
│  2️⃣ إذا تم تمرير sectionId:                                                │
│     • جلب المقطع من قاعدة البيانات                                          │
│     • نسخ التاريخ من Section.date                                           │
│     • نسخ اسم الحلقة من Section.group                                       │
│     • تحديد sessionType تلقائياً:                                           │
│       - إذا يوجد memorizationMeta فقط → "hifz"                              │
│       - إذا يوجد reviewMeta فقط → "murajaah"                                │
│       - إذا كلاهما موجود → "both"                                           │
│                                                                             │
│  3️⃣ التحقق من وجود التاريخ:                                                │
│     • sessionDate مطلوب!                                                    │
│     • يُشتق day تلقائياً من التاريخ                                         │
│                                                                             │
│  4️⃣ فحص التعارض:                                                           │
│     • checkTimeConflict({ teacherId, sessionDate, startHour, endHour })     │
│     • إذا يوجد تعارض → إرجاع خطأ 409 Conflict                               │
│                                                                             │
│  5️⃣ إنشاء الموعد:                                                          │
│     • حفظ في TimeTable                                                      │
│     • isRecurring = false (دائماً)                                          │
│                                                                             │
│  6️⃣ تحديث المقطع (إذا موجود):                                              │
│     • Section.timetableId = timetable._id                                   │
│     • Section.hasSchedule = true                                            │
│     • Section.scheduleStatus = "scheduled"                                  │
│     • Section.scheduleInfo = { day, startHour, endHour }                    │
│                                                                             │
│  7️⃣ إرجاع النتيجة مع populate:                                             │
│     • teacherId → firstName, lastName                                       │
│     • groupId → name                                                        │
│     • sectionId → date, group, memorizationSection, reviewSection           │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                TimeTable.pre("save") → Section Sync                         │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  عند حفظ TimeTable مع sectionId:                                            │
│  ─────────────────────────────                                              │
│  1. نسخ sectionInfo (memorizationSection, reviewSection, marksStatus)       │
│  2. نسخ sessionDate من Section.date                                         │
│  3. نسخ groupId من Section.groupId                                          │
│  4. نسخ note من Section.group                                               │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                TimeTable.post("save") → Section Update                      │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  بعد حفظ TimeTable:                                                         │
│  ─────────────────                                                          │
│  Section.hasSchedule = true                                                 │
│  Section.scheduleStatus = "scheduled"                                       │
│  Section.timetableId = TimeTable._id                                        │
│  Section.scheduleInfo = { day, startHour, endHour }                         │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│               Section.post("save") → TimeTable Update                       │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  بعد حفظ Section مع timetableId:                                            │
│  ────────────────────────────                                               │
│  TimeTable.sessionDate = Section.date                                       │
│  TimeTable.groupId = Section.groupId                                        │
│  TimeTable.note = Section.group                                             │
│  TimeTable.sectionInfo = { memorizationSection, reviewSection, marksStatus }│
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘

┌─────────────────────────────────────────────────────────────────────────────┐
│                         الحذف التلقائي                                      │

│                                                                             │
│  عند حذف TimeTable:                                                         │
│  → Section.hasSchedule = false                                              │

│  → Section.timetableId = null                                               │

                    الجزء العاشر: دوال المساعدة للوقت
================================================================================

📁 ملف المساعد: Backend/src/controllers/TimeTableController/helpers/dateTime.helper.js

┌─────────────────────────────────────────────────────────────────────────────┐
│                          الدوال المتاحة                                     │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  isSummerTime()                                                             │
│  ─────────────                                                              │
│  → true إذا الشهر بين مايو وسبتمبر                                          │
│  → false غير ذلك                                                            │
│                                                                             │
│  generateAvailableHours()                                                   │
│  ────────────────────────                                                   │
│  → Array of hours based on season                                           │
│  → صيفي: ["12:00 PM", "12:30 PM", ..., "9:00 PM"]                           │
│  → شتوي: ["11:00 AM", "11:30 AM", ..., "8:00 PM"]                           │
│                                                                             │
│  timeToMinutes(timeStr)                                                     │
│  ──────────────────────                                                     │
│  → تحويل "2:30 PM" إلى دقائق (14*60 + 30 = 870)                             │
│  → يستخدم للمقارنة الزمنية                                                  │
│                                                                             │
│  getArabicDayFromDate(date)                                                 │
│  ──────────────────────────                                                 │
│  → استخراج اسم اليوم بالعربية من التاريخ                                    │
│  → مثال: new Date("2026-01-12") → "الاثنين"                                 │
│                                                                             │
│  getWeekRange(date)                                                         │
│  ──────────────────                                                         │
│  → حساب بداية ونهاية الأسبوع (السبت - الجمعة)                               │
│  → { startOfWeek, endOfWeek }                                               │
│                                                                             │
│  extractDayInfo(date)                                                       │
│  ────────────────────                                                       │
│  → استخراج معلومات اليوم الكاملة                                            │
│  → { dayName, dayIndex, dateFormatted, dateShort }                          │
│  → مثال: { dayName: "الاثنين", dateFormatted: "12 يناير 2026" }             │
│                                                                             │
│  formatDateArabic(date)                                                     │
│  ──────────────────────                                                     │
│  → "12 يناير 2026"                                                          │
│                                                                             │
│  formatDateShort(date)                                                      │
│  ─────────────────────                                                      │
│  → "12/1/2026"                                                              │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
